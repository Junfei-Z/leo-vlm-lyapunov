% ---- Battery evolution (replaces Eq. 20): platform load drawn every slot ----
\begin{equation}
b_s^{t+1} = b_s^t + \omega_s^t
   - \sum_{i,m} z_{ism}^t \frac{\mathbb{E}[E_{im}]}{N_{im}}
   - P_{\text{base}}\,\tau, \qquad \forall s,t.
\end{equation}

% ---- Energy feasibility (replaces Eq. 21): current slot must survive platform + inference ----
\begin{equation}
\sum_{i,m} z_{ism}^t \frac{\mathbb{E}[E_{im}]}{N_{im}} + P_{\text{base}}\,\tau
   \;\le\; b_s^t + \omega_s^t, \qquad \forall s,t.
\end{equation}

% ---- Energy virtual queue (replaces Eq. 28): back-pressure includes platform load ----
\begin{equation}
Q_s^E(t+1) = \max\!\Big\{\, Q_s^E(t)
   + \sum_{i,m} z_{ism}^t \frac{\mathbb{E}[E_{im}]}{N_{im}}
   + P_{\text{base}}\,\tau - \omega_s^t,\; 0 \,\Big\}.
\end{equation}
